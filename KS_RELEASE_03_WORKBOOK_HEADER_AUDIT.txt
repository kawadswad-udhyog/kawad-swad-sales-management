To unblock Stage C and close both (a) the 1-tab 255-vs-256 delta and (b) all 36 header checks,

  function ksExportReadOnlyWorkbookHeadersForAudit() {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheets = ss.getSheets();
    var out = {
      exported_at_utc: new Date().toISOString(),
      tab_count: sheets.length,
      tabs: []
    };
    for (var i = 0; i < sheets.length; i++) {
      var sh = sheets[i];
      var lastCol = sh.getLastColumn();
      var headers = lastCol > 0 ? sh.getRange(1, 1, 1, lastCol).getValues()[0].map(String) : [];
      out.tabs.push({ name: sh.getName(), row_count: sh.getLastRow(), col_count: lastCol, row1_headers: headers });
    }
    Logger.log(JSON.stringify(out));
  }
