  var physicalByName = {};
  var allTabs = [];
  var deprecatedOrScratchFound = [];

  for (var i = 0; i < sheets.length; i++) {
    var sh = sheets[i];
    var name = sh.getName();
    var lastRow = sh.getLastRow();
    var lastCol = sh.getLastColumn();
    var rawRow1 = lastCol > 0 ? sh.getRange(1, 1, 1, lastCol).getValues()[0] : [];

    var headersPreservedOrder = [];
    var seen = {};
    var duplicateHeaders = [];
    var blankHeaderColumns = [];

    for (var c = 0; c < rawRow1.length; c++) {
      var h = String(rawRow1[c] === null || rawRow1[c] === undefined ? '' : rawRow1[c]).trim();
      headersPreservedOrder.push(h);
      if (!h) {
        blankHeaderColumns.push(c + 1);
      } else {
        var key = h.toLowerCase();
        if (Object.prototype.hasOwnProperty.call(seen, key)) {
          duplicateHeaders.push({ header: h, col_index_1based: c + 1, first_col_1based: seen[key] });
        } else {
          seen[key] = c + 1;
        }
      }
    }

    var entry = {
      sheet_index_1based: i + 1,
      sheet_name: name,
      is_hidden: sh.isSheetHidden(),
      total_rows_including_header: lastRow,
      data_row_count: lastRow > 0 ? (lastRow - 1) : 0,
      col_count: lastCol,
      row1_headers_ordered: headersPreservedOrder,
      duplicate_headers: duplicateHeaders,
      blank_header_columns: blankHeaderColumns
    };

    allTabs.push(entry);
    physicalByName[name] = entry;

    if (name === 'ExceptionLogs' || (scratchRe.test(name) && name !== 'DBV5_FIELD_DICTIONARY_3074')) {
      deprecatedOrScratchFound.push({
        sheet_name: name,
        data_row_count: entry.data_row_count,
        reason: name === 'ExceptionLogs' ? 'DEPRECATED_LEGACY_ALIAS' : 'FORBIDDEN_SCRATCH_PREFIX'
      });
    }
  }

  // Audit all 36 guarded tabs (12 CONFLICTING_CORE_SCHEMA_TABS + 24 PENDING_OWNER_MAPPING_TABS)
  var guardedAudit = [];
  var guardedNames = Object.keys(KS_R05_GUARDED_TABS_EXPECTED).sort();
  for (var g = 0; g < guardedNames.length; g++) {
    var tabName = guardedNames[g];
    var spec = KS_R05_GUARDED_TABS_EXPECTED[tabName];
    var phys = physicalByName[tabName] || null;
    var actual = phys ? phys.row1_headers_ordered : [];
    var actualSet = {};
    for (var a = 0; a < actual.length; a++) {
      if (actual[a]) actualSet[actual[a]] = true;
    }

    function diffAgainst(expectedList) {
      if (!expectedList || expectedList.length === 0) {
        return { expected_count: 0, missing_from_live: [], unexpected_in_live: actual.slice(), exact_order_match: false };
      }
      var expSet = {};
      var missing = [];
      for (var x = 0; x < expectedList.length; x++) {
        expSet[expectedList[x]] = true;
        if (!actualSet[expectedList[x]]) missing.push(expectedList[x]);
      }
      var unexpected = [];
      for (var y = 0; y < actual.length; y++) {
        if (actual[y] && !expSet[actual[y]]) unexpected.push(actual[y]);
      }
      var orderMatch = (actual.length === expectedList.length);
      if (orderMatch) {
        for (var z = 0; z < expectedList.length; z++) {
          if (actual[z] !== expectedList[z]) { orderMatch = false; break; }
        }
      }
      return {
        expected_count: expectedList.length,
        missing_from_live: missing,
        unexpected_in_live: unexpected,
        exact_order_match: orderMatch
      };
    }

    guardedAudit.push({
      tab_name: tabName,
      guard_group: spec.guard_group,
      physically_present: Boolean(phys),
      data_row_count: phys ? phys.data_row_count : null,
      actual_headers_ordered: actual,
      duplicate_headers: phys ? phys.duplicate_headers : [],
      blank_header_columns: phys ? phys.blank_header_columns : [],
      diff_vs_required_tabs: diffAgainst(spec.expected_required_tabs),
      diff_vs_blueprint_v2: diffAgainst(spec.expected_blueprint_v2)
    });
  }

  // Bootstrap pre-check (counts ONLY, zero user data)
  var usersTab = physicalByName['Users'] || null;
  var authUsersTab = physicalByName['AuthUsers'] || null;

  return {
    schema_export_version: 'ks-release-05-readonly-audit-v1',
    read_only_guarantee: 'ROW_1_HEADERS_AND_COUNTS_ONLY_ZERO_WRITES_ZERO_BUSINESS_DATA',
    exported_at_utc: new Date().toISOString(),
    workbook_metadata: {
      workbook_name: ss.getName(),
      spreadsheet_id_redacted: String(ss.getId()).slice(0, 4) + '****' + String(ss.getId()).slice(-4),
      timezone: ss.getSpreadsheetTimeZone(),
      locale: ss.getSpreadsheetLocale(),
      physical_sheet_count: sheets.length
    },
    bootstrap_precheck_counts_only: {
      Users_present: Boolean(usersTab),
      Users_data_row_count: usersTab ? usersTab.data_row_count : null,
      Users_row1_headers: usersTab ? usersTab.row1_headers_ordered : [],
      AuthUsers_present: Boolean(authUsersTab),
      AuthUsers_data_row_count: authUsersTab ? authUsersTab.data_row_count : null,
      AuthUsers_row1_headers: authUsersTab ? authUsersTab.row1_headers_ordered : []
    },
    deprecated_or_scratch_tabs_found: deprecatedOrScratchFound,
    guarded_36_tabs_audit: guardedAudit,
    all_physical_tabs: allTabs
  };
}

/**
 * Entry Point A (Execution Log — zero UI permissions needed).
 * Splits JSON cleanly across 15,000-char chunks so Apps Script Logger never truncates output.
 */
function ksExportReadOnlyWorkbookAuditV5() {
  var report = ksBuildReadOnlyWorkbookAuditReportV5_();
  var jsonText = JSON.stringify(report, null, 2);
  var chunkSize = 15000;
  var totalChunks = Math.ceil(jsonText.length / chunkSize);
  Logger.log('=== KS_AUDIT_JSON_START (total_chars=' + jsonText.length + ', chunks=' + totalChunks + ') ===');
  for (var i = 0; i < totalChunks; i++) {
    Logger.log('--- CHUNK ' + (i + 1) + '/' + totalChunks + ' ---\n' + jsonText.slice(i * chunkSize, (i + 1) * chunkSize));
  }
  Logger.log('=== KS_AUDIT_JSON_END ===');
  return 'OK: Exported ' + report.workbook_metadata.physical_sheet_count + ' sheets (read-only).';
}
