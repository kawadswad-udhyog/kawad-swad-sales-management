```javascript
function ksExportReadOnlyWorkbookHeadersForAudit() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheets = ss.getSheets();
  var out = {
    schema_export_version: 'ks-release-04-readonly-v1',
    exported_at_utc: new Date().toISOString(),
    tab_count: sheets.length,
    tabs: []
  };
  for (var i = 0; i < sheets.length; i++) {
    var sh = sheets[i];
    var lastCol = sh.getLastColumn();
    var row1 = lastCol > 0 ? sh.getRange(1, 1, 1, lastCol).getValues()[0].map(function(v) { return String(v).trim(); }) : [];
    out.tabs.push({
      index: i + 1,
      name: sh.getName(),
      row_count: sh.getLastRow(),
      col_count: lastCol,
      row1_headers: row1
    });
  }
  Logger.log(JSON.stringify(out));
}
```
